# Runtime dependencies — what the deployed service needs and nothing more.
# Development and CI tooling lives in requirements-dev.txt so the production
# image (backend/Dockerfile) doesn't ship a linter.
#
# Every version is pinned exactly so a deploy six months from now installs what
# CI tested. Dependabot (.github/dependabot.yml) proposes the bumps and
# `pip-audit` (.github/workflows/supply-chain.yml) fails the build if a pin has
# a known advisory against it.

# Held at >= 0.141 rather than 0.115 because of what it drags in, not for
# anything in its own changelog: fastapi's starlette range is the only lever on
# starlette's version, and 0.115 caps it below 0.39 — eight PYSEC advisories
# short of a fix. 0.141 allows starlette 1.x, which clears all eight.
fastapi==0.141.1
uvicorn[standard]==0.32.0
pydantic==2.9.2
# uvicorn[standard] pulls this in for the /ws protocol; pinned explicitly
# because the WebSocket stream is a first-class feature, not an accident.
websockets==13.1

# python-dotenv was here but nothing imports it: config.py reads os.getenv
# directly and no code calls load_dotenv(). Rather than wire up a .env loader
# nobody asked for, the docs now say plainly that configuration comes from real
# environment variables — see the Configuration table in README.md.

# Persistence
SQLAlchemy==2.0.35
alembic==1.13.3

# Used by db/seed.py, which runs on startup to populate an empty database.
Faker==30.3.0
